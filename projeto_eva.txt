import json, os, others, weight, random

try:
    import readline
    hist_path = os.path.expanduser("~/.eva_history")
    if os.path.exists(hist_path): readline.read_history_file(hist_path)
except: pass

class EvaCore:
    def __init__(self):
        os.system('clear' if os.name == 'posix' else 'cls')
        self.mem_path = 'learning.json'
        self.words_path = 'words.txt'
        self.memoria = self.carregar()
        self.last_key = None
        self.modo_ensino = False

    def carregar(self):
        if os.path.exists(self.mem_path):
            with open(self.mem_path, 'r', encoding='utf-8') as f:
                try:
                    data = json.load(f)
                    return data if isinstance(data, dict) else {}
                except: return {}
        return {}

    def obter_social(self, tag):
        frases = []
        if os.path.exists(self.words_path):
            with open(self.words_path, 'r', encoding='utf-8') as f:
                cap = False
                for l in f:
                    l = l.strip()
                    if l == f"[{tag}]": cap = True; continue
                    if l.startswith("[") and cap: break
                    if cap and l: frases.append(l)
        return random.choice(frases) if frases else None

    def salvar(self):
        with open(self.mem_path, 'w', encoding='utf-8') as f:
            json.dump(self.memoria, f, indent=4, ensure_ascii=False)
            try:
                import readline
                readline.write_history_file(os.path.expanduser("~/.eva_history"))
            except: pass

    def escutar(self):
        print("\033[96m" + "="*40 + "\n  EVA - SISTEMAS DE INTELIGÊNCIA ATIVOS\n" + "="*40 + "\033[0m")
        while True:
            try:
                user = input("\033[92mVocê:\033[0m ").strip()
                if not user: continue
                u_low = user.lower()

                # --- 1. MODO ENSINO ---
                if self.modo_ensino:
                    if u_low == 'cancelar':
                        self.modo_ensino = False; print("Eva: Aula cancelada."); continue
                    if self.last_key:
                        self.memoria[self.last_key] = {"texto": user, "peso": 10}
                        self.salvar(); self.modo_ensino = False; print("Eva: Entendido!")
                    continue

                # --- 2. COMANDOS DE SISTEMA (SISTEMA DE PESOS) ---
                if u_low == '$ensinar':
                    self.modo_ensino = True; print(f"Eva: O que é '{self.last_key}'?"); continue

                if u_low == '$ng' and self.last_key:
                    suc, msg, critico = weight.ajustar_peso(self.memoria, self.last_key, -2)
                    self.salvar()
                    print(f"\033[91mSystem: {msg}\033[0m")
                    if critico:
                        res = others.roteador(self.last_key)
                        if res:
                            self.memoria[self.last_key] = {"texto": res, "peso": 1}
                            self.salvar()
                            print(f"\033[94mEva (Novo):\033[0m {res}")
                    continue

                if u_low == '$p' and self.last_key:
                    suc, msg, _ = weight.ajustar_peso(self.memoria, self.last_key, 6)
                    self.salvar()
                    print(f"\033[92mSystem: {msg}\033[0m")
                    continue

                if u_low == '$g' and self.last_key:
                    suc, msg, _ = weight.ajustar_peso(self.memoria, self.last_key, 2)
                    self.salvar()
                    print(f"\033[93mSystem: {msg}\033[0m")
                    continue

                # --- 3. PROCESSAMENTO DE DIÁLOGO ---
                alvo = others.extrair_alvo(user)
                if not alvo: alvo = user
                intencao = others.identificar_intencao(user)

                # Atualiza a última chave apenas se não for um comando
                if u_low not in ['$g', '$p', '$ng']: self.last_key = alvo

                # Verifica Memória (Rank > 0)
                if weight.validar_memoria(self.memoria, self.last_key):
                    m = self.memoria[self.last_key]
                    print(f"\033[94mEva (Memória [R:{m['peso']}]):\033[0m {m['texto']}")
                else:
                    # Busca Web
                    print(f"\033[90mEva: Analisando '{self.last_key}' (Tipo: {intencao})...\033[0m")
                    res = others.roteador(user)
                    if res:
                        self.memoria[self.last_key] = {"texto": res, "peso": 1}
                        self.salvar()
                        print(f"\033[94mEva (Novo):\033[0m {res}")
                    else:
                        social = self.obter_social("ERRO_BUSCA")
                        print(f"\033[94mEva:\033[0m {social or 'Ainda não conheço isso.'}")

            except KeyboardInterrupt: break
            except Exception as e: print(f"Erro no Sistema: {e}")

if __name__ == "__main__":
    EvaCore().escutar()

import re, defs, use, locates

def identificar_intencao(frase):
    f = frase.lower()
    # LOCALIZAÇÃO (Módulo locates)
    if re.search(r'\bonde\b|\bfica\b|\bencontrar\b|\blocalização\b|\blocalizar\b', f): 
        return 'localizacao'
    
    # INSTRUÇÃO / UTILIDADE (Módulo use)
    if re.search(r'\bcomo\b|\bpasso a passo\b|\busar\b|\bservir\b|\bpra que serve\b|\bfunção\b', f): 
        return 'instrucao'
    
    # ORIGEM (Tratado pelo defs)
    if re.search(r'\bquem\b|\bcriador\b|\borigem\b|\bhistória\b', f): 
        return 'origem'
    
    return 'definicao'

def extrair_alvo(frase):
    # Remove pontuação básica e limpa espaços
    alvo = re.sub(r'[?!.,]', '', frase.lower()).strip()
    
    # Lista de prefixos para limpar (Organizada para não deixar resíduos)
    termos_sujeira = [
        r'^o que (são|sao|é|e)\s+(o|a|os|as)\s+', 
        r'^o que (são|sao|é|e)\s+',
        r'^como (usar|funciona|fazer|serve)\s+(o|a|os|as|um|uma)\s+',
        r'^como (usar|funciona|fazer|serve)\s+',
        r'^pra que serve (o|a|os|as|um|uma)\s+',
        r'^pra que serve\s+',
        r'^onde (encontrar|fica|localizar)\s+(o|a|os|as|um|uma)\s+',
        r'^onde (encontrar|fica|localizar)\s+',
        r'^(o|a|os|as|um|uma)\s+'
    ]
    
    for pat in termos_sujeira:
        alvo = re.sub(pat, '', alvo).strip()
    
    return alvo

def roteador(frase):
    # 1. Identifica a intenção
    intencao = identificar_intencao(frase)
    
    # 2. Extrai o alvo (limpa a frase)
    alvo = extrair_alvo(frase)
    
    # Se a limpeza falhar e o alvo ficar vazio, usa a frase original
    if not alvo or len(alvo) < 2:
        alvo = frase

    # DEBUG para monitorar no terminal
    print(f"\033[90m[Debug] Intenção: {intencao} | Alvo: {alvo}\033[0m")

    # 3. Encaminha para o módulo especialista
    if intencao == 'instrucao': 
        return use.buscar(alvo)
    
    if intencao == 'localizacao':
        return locates.buscar(alvo)
    
    # Padrão: Definições e Origem (Módulo defs)
    return defs.buscar(alvo, intencao)

def ajustar_peso(memoria, chave, valor):
    # Verifica se a chave existe e se é um dicionário (evita erro de 'str')
    if chave not in memoria or not isinstance(memoria[chave], dict): 
        return False, "Alvo inválido ou não encontrado", False
    
    # Garante que a chave 'peso' existe antes de somar
    if 'peso' not in memoria[chave]:
        memoria[chave]['peso'] = 1
        
    memoria[chave]['peso'] += valor
    np = memoria[chave]['peso']
    
    # Se o peso cair para 0 ou menos, sinaliza para o brain buscar outro
    if np <= 0: 
        return True, f"Rank crítico ({np}). Vou buscar outro.", True
    
    return True, f"Rank atualizado: {np}", False

def validar_memoria(memoria, chave):
    # Só valida se: existir, for dicionário, tiver a chave 'peso' e o peso for > 0
    if chave in memoria and isinstance(memoria[chave], dict):
        return memoria[chave].get('peso', 0) > 0
    return False

                    return txt
    except: pass
    return None

import requests
from bs4 import BeautifulSoup
import re

def executar_busca(termo, intencao='geral'):
    headers = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/119.0.0.0 Safari/537.36'}
    
    queries = {
        'origem': [f"quem criou {termo}", f"inventor de {termo}"],
        'localizacao': [f"onde encontrar {termo}", f"onde nasce {termo}"],
        'instrucao': [f"como funciona {termo}"],
        'definicao': [f"o que é {termo}"]
    }
    
    lista = queries.get(intencao, [termo])
    
    for q in lista:
        try:
            r = requests.get(f"https://www.google.com/search?q={q.replace(' ', '+')}&hl=pt-BR", headers=headers, timeout=5)
            if r.status_code == 200:
                soup = BeautifulSoup(r.text, 'html.parser')
                # Seletores universais do Google para respostas rápidas
                for sel in ['.BNeawe', '.s3v9rd', '.AP7Wnd', '.VwiC3b']:
                    for res in soup.select(sel):
                        txt = res.get_text().strip()
                        if len(txt) > 25: return limpar(txt)
        except: continue

        try:
            wiki = f"https://pt.wikipedia.org/w/api.php?action=opensearch&search={q.replace(' ', '+')}&limit=1&format=json"
            s = requests.get(wiki, headers=headers, timeout=3).json()
            if s[1]:
                t = s[1][0].replace(' ', '_')
                res = requests.get(f"https://pt.wikipedia.org/api/rest_v1/page/summary/{t}", headers=headers, timeout=3).json()
                if res.get('extract'): return limpar(res['extract'])
        except: continue
    return None

def limpar(txt):
    if not txt: return None
    txt = re.sub(r'\[.*?\]', '', txt)
    txt = " ".join(txt.split())
    p = txt.split()
    return " ".join(p[:60]) + ("..." if len(p) > 60 else "")
