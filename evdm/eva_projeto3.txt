import os
import getpass

SENHA = "hatepass"

senha = getpass.getpass("Senha: ")

if senha != SENHA:
    print("Senha incorreta.")
    exit()

print("Salvando no GitHub...")

# pasta do projeto
PASTA = "/data/data/com.termux/files/home/eva"

os.chdir(PASTA)

# comandos git
os.system("git add .")
os.system('git commit -m "AutoSave EVA"')
os.system("git push origin main")

print("Backup enviado.")
